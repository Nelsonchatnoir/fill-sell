## Annexe — la requête du parc (§ 2), lecture seule

Les fonctions de preuve (retrait_job_prouve, rattachement_confirme) sont réservées au serveur : leur logique est recopiée en ligne (`prouve`, `confirme_approx`).

```sql
WITH vend AS MATERIALIZED (
  SELECT v.inventaire_id, count(*) n_ventes,
         min(vente_instant(v.vendu_le, v.date, v.created_at)) vendu_le,
         (array_agg(lower(coalesce(nullif(v.plateforme_code,''), nullif(v.plateforme,''), '')) ORDER BY v.created_at))[1] vendu_sur,
         (array_agg(v.id ORDER BY v.created_at))[1] vente_id,
         (array_agg(v.created_at ORDER BY v.created_at))[1] vente_cree,
         (array_agg(v.source ORDER BY v.created_at))[1] vente_source,
         (array_agg(v.annonce_id ORDER BY v.created_at))[1] vente_annonce,
         (array_agg(v.commande_ref ORDER BY v.created_at))[1] vente_cmd
    FROM ventes v
   WHERE v.inventaire_id IS NOT NULL AND lower(coalesce(v.statut,'')) NOT IN ('annule','annulée','cancelled','canceled')
   GROUP BY 1),
cand AS MATERIALIZED (
  SELECT i.id fiche, i.user_id, i.titre ftitre, i.quantite fq, i.plateforme fpl, i.vinted_item_id fvinted, vd.*,
         a.id annonce, a.platform, a.listing_id, a.url, a.titre, a.prix, a.vu_le, a.created_at a_cree, a.publiee_le, a.quantite aq, a.job_id, a.source_rapprochement src, a.statut_plateforme st, a.transaction_plateforme tx
    FROM inventaire i JOIN vend vd ON vd.inventaire_id = i.id
    JOIN annonces_plateforme a ON a.inventaire_id = i.id AND a.user_id = i.user_id
   WHERE i.statut = 'vendu' AND i.fusionne_dans IS NULL
     AND a.disparu_le IS NULL AND a.retiree_le IS NULL AND a.ignoree_le IS NULL
     AND a.statut_plateforme IN ('en_ligne','en_verification')
     AND a.vu_le > vd.vendu_le
     AND a.platform <> vd.vendu_sur),
jx AS MATERIALIZED (
  SELECT j.*, coalesce(j.platform_fields #>> '{rattachement,par}','') par, coalesce(j.platform_fields #>> '{rattachement,motif}','') motif,
         coalesce(j.platform_fields #>> '{rattachement,import}','') imp, coalesce(j.platform_fields ->> 'source','') src_j,
         EXISTS (SELECT 1 FROM inventaire_fusions f WHERE f.defait_le IS NULL AND coalesce(f.par,'') NOT LIKE 'utilisateur%'
                   AND coalesce(f.par,'') NOT LIKE '%photo\_identique)' AND jsonb_typeof(f.deplacements->'cross_post_jobs')='array'
                   AND (f.deplacements->'cross_post_jobs') ? j.id::text) fusion_auto
    FROM cross_post_jobs j WHERE j.id IN (SELECT job_id FROM cand))
SELECT jsonb_agg(jsonb_build_object(
  'compte', split_part(p.email,'@',1), 'user', c.user_id, 'fiche', c.fiche::text, 'ftitre', c.ftitre, 'fpl', c.fpl, 'vendu_sur', c.vendu_sur,
  'vendu_le', c.vendu_le, 'vente', c.vente_id, 'vente_cree', c.vente_cree, 'vente_src', c.vente_source, 'vente_annonce', c.vente_annonce, 'vente_cmd', c.vente_cmd,
  'annonce', c.annonce, 'platform', c.platform, 'listing', c.listing_id, 'url', c.url, 'titre', c.titre, 'prix', c.prix, 'vu', c.vu_le,
  'a_cree', c.a_cree, 'publiee', c.publiee_le, 'src', c.src, 'st', c.st, 'tx', c.tx,
  'derniere_activite', (SELECT max(x.vu_le) FROM annonces_plateforme x WHERE x.user_id = c.user_id AND x.platform = c.platform),
  'vivantes_pl', (SELECT count(DISTINCT ap.listing_id) FROM annonces_plateforme ap WHERE ap.inventaire_id = c.fiche AND ap.platform = c.platform
                    AND ap.disparu_le IS NULL AND ap.retiree_le IS NULL AND ap.ignoree_le IS NULL AND ap.statut_plateforme IN ('en_ligne','en_verification')),
  'job', (SELECT jsonb_build_object('id', j.id, 'action', j.action, 'status', j.status, 'listing', j.platform_listing_id, 'url', j.listing_url,
      'pub', j.published_at, 'cree', j.created_at, 'src', j.src_j, 'ratt', j.platform_fields->'rattachement',
      'sale_signal', j.platform_fields->>'sale_signal', 'err', left(j.error, 160), 'vu_en_ligne', j.vu_en_ligne_le,
      'prouve', (j.par='utilisateur' AND j.motif='meme_article_confirme') OR (NOT j.fusion_auto AND (j.par IN ('','utilisateur') OR j.motif IN ('identifiant_depot_clos','photo_identique') OR j.imp='true')
                 AND (j.src_j <> 'releve' OR j.par='utilisateur' OR j.motif IN ('identifiant_depot_clos','photo_identique') OR j.imp='true')),
      'confirme_approx', (j.src_j <> 'releve' AND (j.platform_fields->'rattachement') IS NULL) OR (j.par <> 'auto' OR j.imp='true' OR j.motif IN ('identifiant_depot_clos','photo_identique')),
      'fusion_auto', j.fusion_auto,
      'importee', j.src_j = 'releve' OR j.imp = 'true', 'retrait_arme', j.platform_fields->'retrait_arme', 'pending_removal', j.platform_fields->>'pending_removal', 'reporte', j.platform_fields->>'pending_removal_reporte_le', 'vendu_encore', j.platform_fields->'vendu_encore_en_ligne', 'eteint', j.platform_fields->'pending_removal_eteint')
      FROM jx j WHERE j.id = c.job_id),
  'jobs_publies', (SELECT jsonb_agg(jsonb_build_object('id', j.id, 'listing', j.platform_listing_id, 'url', j.listing_url, 'action', j.action))
      FROM cross_post_jobs j WHERE j.inventaire_id = c.fiche AND j.platform = c.platform AND j.status = 'published' AND coalesce(j.action,'publish') IN ('publish','republish')),
  'deletes', (SELECT jsonb_agg(jsonb_build_object('id', d.id, 'st', d.status, 'cree', d.created_at, 'listing', d.platform_listing_id, 'url', d.listing_url, 'err', left(d.error,160), 'chemin', d.platform_fields#>>'{arme_par,chemin}', 'deja_absente', d.platform_fields->'deja_absente') ORDER BY d.created_at)
      FROM cross_post_jobs d WHERE d.inventaire_id = c.fiche AND d.platform = c.platform AND d.action = 'delete'),
  'questions', (SELECT jsonb_agg(jsonb_build_object('id', q.id, 'st', q.statut, 'motif', q.motif, 'cree', q.created_at, 'job', q.preuves->>'job', 'annonce', q.preuves->>'annonce_id', 'decide', q.decide_le, 'src', q.source, 'issue', q.preuves->>'issue'))
      FROM inventaire_doublons q WHERE q.user_id = c.user_id AND (q.garde = c.fiche OR q.absorbe = c.fiche)),
  'serie_annonce', (SELECT count(*) FROM annonces_plateforme x WHERE x.user_id = c.user_id AND x.platform = c.platform AND x.disparu_le IS NULL AND x.ignoree_le IS NULL AND rapprochement_texte_norm(x.titre) = rapprochement_texte_norm(c.titre)),
  'serie_fiche', (SELECT max(n) FROM (SELECT count(*) n FROM annonces_plateforme x WHERE x.user_id = c.user_id AND x.disparu_le IS NULL AND x.ignoree_le IS NULL AND rapprochement_texte_norm(x.titre) = rapprochement_texte_norm(c.ftitre) GROUP BY x.platform) s),
  'n_ventes', c.n_ventes, 'fq', c.fq, 'aq', c.aq
) ORDER BY split_part(p.email,'@',1), c.fiche) r
FROM cand c LEFT JOIN profiles p ON p.id = c.user_id;
```
